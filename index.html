function montarBaseVariavel(registros){ const mapa = {}; registros.forEach(r => { if (!r.cpf || !r.mes || !r.item) return; // ignora linhas incompletas const cpfLimpo = limparCpf(r.cpf); if (!mapa[cpfLimpo]) { mapa[cpfLimpo] = { nome: r.nome, cpf: r.cpf, funcao: r.funcao, turno: r.turno, meses: {} }; } const mes = (r.mes || '').trim().toUpperCase(); if (!mapa[cpfLimpo].meses[mes]) mapa[cpfLimpo].meses[mes] = { itens: {}, total: 0 }; const valorNum = parseFloat((r.valor || '0').replace(',', '.')) || 0; const itemNome = (r.item || '').trim(); if (itemNome.toUpperCase() === 'TOTAL') { mapa[cpfLimpo].meses[mes].total = valorNum; } else if (itemNome) { mapa[cpfLimpo].meses[mes].itens[itemNome] = valorNum; } }); return mapa; }
